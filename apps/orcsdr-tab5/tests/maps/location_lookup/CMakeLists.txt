cmake_minimum_required(VERSION 3.16)
project(orcsdr_location_lookup_tests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CJSON_ROOT "" CACHE PATH "Existing cJSON source directory")
if(NOT EXISTS "${CJSON_ROOT}/cJSON.c")
  message(FATAL_ERROR "Set CJSON_ROOT to the existing SDK cJSON source")
endif()
add_executable(location_lookup_tests location_lookup_tests.cpp
  ../../../ui/location_estimate.cpp "${CJSON_ROOT}/cJSON.c")
target_include_directories(location_lookup_tests PRIVATE stubs ../http_stubs ../../../ui "${CJSON_ROOT}")
if(MSVC)
  target_compile_options(location_lookup_tests PRIVATE /UNDEBUG)
  target_compile_definitions(location_lookup_tests PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
  target_compile_options(location_lookup_tests PRIVATE -UNDEBUG)
endif()
enable_testing()
add_test(NAME location_lookup COMMAND location_lookup_tests)
